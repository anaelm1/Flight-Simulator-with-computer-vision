 x = [0; 0; 10]; xdot = zeros(3, 1); theta = zeros(3, 1);

 % The magnitude of the deviation is in radians / second.
 deviation = 100; thetadot = deg2rad(2 * deviation * rand(3, 1) - deviation);
 % Step through the simulation, updating the state.
 for t = times
 % Take input from our controller.
 i = input(t); omega = thetadot2omega(thetadot, theta);
 % Compute linear and angular accelerations.
 a = acceleration(i, theta, xdot, m, g, k, kd); omegadot = angular_acceleration(i, omega, I, L, b, k); omega = omega + dt * omegadot; thetadot = omega2thetadot(omega, theta); theta = theta + dt * thetadot; xdot = xdot + dt * a; x = x + dt * xdot; end
